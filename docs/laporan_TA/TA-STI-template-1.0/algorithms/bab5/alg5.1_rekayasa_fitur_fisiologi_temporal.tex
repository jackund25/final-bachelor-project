\begin{algorithm}[h]
    \caption{Rekayasa Fitur Berbasis Fisiologi dan Temporal}
    \label{alg:alg5.1_rekayasa_fitur}
    \begin{algorithmic}[1]
        \Require Data logbook per pasien yang telah divalidasi
        \Ensure Vektor fitur fisiologis dan temporal
        \State Urutkan data berdasarkan \texttt{patient\_id} dan \texttt{timestamp}
        \ForAll{pasien}
            \State Hitung selang waktu antar-pengamatan
            \State Hitung \texttt{glucose\_delta} dan \texttt{glucose\_rate}
            \State Hitung peluruhan IOB berdasarkan selang waktu dan $\tau_{IOB}$
            \State Hitung peluruhan COB berdasarkan selang waktu dan $\tau_{COB}$
            \State Hitung \texttt{activity}
            \State Hitung \texttt{hour\_sin} dan \texttt{hour\_cos}
            \State Hitung \texttt{time\_since\_prev\_glucose}
            \State Bentuk vektor sembilan fitur
        \EndFor
        \State \Return seluruh vektor fitur
    \end{algorithmic}
\end{algorithm}
